package com.handson.client;

import net.minecraft.entity.item.ItemEntity;

/** Client-side state of the two hands. Index 0 = left hand, 1 = right hand. */
public class HandsState {
    public static boolean modeOn = false;
    public static boolean locked = false;

    public static final boolean[] down = new boolean[2];
    public static final float[] ext = new float[2];
    public static final float[] prevExt = new float[2];
    public static final ItemEntity[] target = new ItemEntity[2];

    // ring inventory around the player
    public static boolean invOpen = false;
    public static float anchorYaw = 0.0F;
    public static final int[] ringHover = {-1, -1};

    // smoothed head rotation, used for the lag of the hands
    public static boolean smoothInit = false;
    public static float smoothYaw = 0.0F;
    public static float smoothPitch = 0.0F;
    public static float prevSmoothYaw = 0.0F;
    public static float prevSmoothPitch = 0.0F;

    public static void reset() {
        modeOn = false;
        locked = false;
        clearHands();
    }

    public static void clearHands() {
        for (int i = 0; i < 2; i++) {
            down[i] = false;
            ext[i] = 0.0F;
            prevExt[i] = 0.0F;
            target[i] = null;
            ringHover[i] = -1;
        }
        invOpen = false;
        smoothInit = false;
    }
}

package com.handson;

import net.minecraft.entity.player.PlayerEntity;
import net.minecraft.inventory.Inventory;
import net.minecraft.item.ItemStack;
import net.minecraft.nbt.CompoundNBT;

/** Storage of the slots of the ring inventory. On the server it is saved in the persistent data of the player. */
public class RingInventory extends Inventory {
    public static final String TAG = "HandsOnRing";

    private final PlayerEntity owner;
    private boolean loading = false;

    public RingInventory(PlayerEntity owner) {
        super(HandSlots.RING_COUNT);
        this.owner = owner;
        if (!owner.level.isClientSide) {
            load();
        }
    }

    private void load() {
        CompoundNBT root = owner.getPersistentData();
        if (!root.contains(TAG, 10)) {
            return;
        }
        CompoundNBT tag = root.getCompound(TAG);
        loading = true;
        for (int i = 0; i < getContainerSize(); i++) {
            setItem(i, ItemStack.of(tag.getCompound("S" + i)));
        }
        loading = false;
    }

    @Override
    public void setChanged() {
        super.setChanged();
        if (loading || owner.level.isClientSide) {
            return;
        }
        CompoundNBT tag = new CompoundNBT();
        for (int i = 0; i < getContainerSize(); i++) {
            tag.put("S" + i, getItem(i).save(new CompoundNBT()));
        }
        owner.getPersistentData().put(TAG, tag);
    }
}

package com.handson;

import java.lang.reflect.Method;
import net.minecraft.entity.player.PlayerEntity;
import net.minecraft.inventory.container.Container;
import net.minecraft.inventory.container.Slot;
import net.minecraft.item.ItemStack;
import net.minecraftforge.fml.common.ObfuscationReflectionHelper;

/** The two hand slots: extra slots of the player's inventory menu (index 46 = left hand, 47 = right hand). */
public class HandSlots {
    public static final int VANILLA_SLOTS = 46;
    public static final int LEFT = 46;
    public static final int RIGHT = 47;

    /** Ring inventory: slots 0-7 around the player, 8-11 above and 12-15 below the four straight ones. */
    public static final int RING_START = 48;
    public static final int RING_COUNT = 16;
    public static final int SCREEN_RING_X = 180;
    public static final int SCREEN_RING_Y = 82;

    /** Position of the slots in the vanilla inventory screen (relative to the screen, right of the panel). */
    public static final int SCREEN_X = 180;
    public static final int SCREEN_Y_LEFT = 24;
    public static final int SCREEN_Y_RIGHT = 58;

    private static Method addSlot;
    private static boolean failed = false;

    /** Adds the two slots to the inventory menu of the player (the same code runs on client and server). */
    public static void install(PlayerEntity player) {
        if (failed) {
            return;
        }
        Container menu = player.inventoryMenu;
        if (menu.slots.size() != VANILLA_SLOTS) {
            return;
        }
        try {
            if (addSlot == null) {
                addSlot = ObfuscationReflectionHelper.findMethod(Container.class, "func_75146_a", Slot.class);
            }
            HandsInventory inv = new HandsInventory(player);
            addSlot.invoke(menu, new Slot(inv, 0, SCREEN_X, SCREEN_Y_LEFT));
            addSlot.invoke(menu, new Slot(inv, 1, SCREEN_X, SCREEN_Y_RIGHT));
            RingInventory ring = new RingInventory(player);
            for (int i = 0; i < RING_COUNT; i++) {
                addSlot.invoke(menu, new Slot(ring, i, SCREEN_RING_X + (i % 4) * 18, SCREEN_RING_Y + (i / 4) * 18));
            }
        } catch (Exception e) {
            failed = true;
            e.printStackTrace();
        }
    }

    public static boolean ringReady(PlayerEntity player) {
        return player.inventoryMenu.slots.size() >= RING_START + RING_COUNT;
    }

    public static ItemStack getRing(PlayerEntity player, int index) {
        if (!ringReady(player) || index < 0 || index >= RING_COUNT) {
            return ItemStack.EMPTY;
        }
        return player.inventoryMenu.getSlot(RING_START + index).getItem();
    }

    public static void setRing(PlayerEntity player, int index, ItemStack stack) {
        if (!ringReady(player) || index < 0 || index >= RING_COUNT) {
            return;
        }
        player.inventoryMenu.getSlot(RING_START + index).set(stack);
    }

    public static boolean ready(PlayerEntity player) {
        return player.inventoryMenu.slots.size() > RIGHT;
    }

    /** hand: 0 = left, 1 = right. */
    public static ItemStack get(PlayerEntity player, int hand) {
        if (!ready(player)) {
            return ItemStack.EMPTY;
        }
        return player.inventoryMenu.getSlot(LEFT + hand).getItem();
    }

    public static void set(PlayerEntity player, int hand, ItemStack stack) {
        if (!ready(player)) {
            return;
        }
        player.inventoryMenu.getSlot(LEFT + hand).set(stack);
    }
}

package com.handson.net;

import com.handson.HandSlots;
import com.handson.client.ClientHooks;
import com.handson.server.ServerEvents;
import java.util.function.Supplier;
import net.minecraft.entity.Entity;
import net.minecraft.entity.item.ItemEntity;
import net.minecraft.entity.player.ServerPlayerEntity;
import net.minecraft.item.ItemStack;
import net.minecraft.network.PacketBuffer;
import net.minecraft.util.Util;
import net.minecraft.util.math.vector.Vector3d;

import net.minecraftforge.api.distmarker.Dist;
import net.minecraftforge.fml.DistExecutor;
import net.minecraftforge.fml.network.NetworkEvent;

public class HandsPacket {
    /** Client to server: player asks to switch between vanilla and Hands On controls (needs cheats). */
    public static final int REQUEST_TOGGLE = 0;
    /** Client to server: grab the item entity with the given id. */
    public static final int GRAB = 1;
    /** Client to server: drop the item in the given hand. */
    public static final int DROP = 2;
    /** Server to client: set the control mode. */
    public static final int SET_MODE = 3;
    /** Client to server: put the item of the given hand into the ring slot (or take it from the slot). */
    public static final int RING = 4;

    public final int action;
    public final int hand; // 0 = left hand slot, 1 = right hand slot
    public final int id;
    public final boolean flag;

    public HandsPacket(int action, int hand, int id, boolean flag) {
        this.action = action;
        this.hand = hand;
        this.id = id;
        this.flag = flag;
    }

    public static void encode(HandsPacket p, PacketBuffer buf) {
        buf.writeVarInt(p.action);
        buf.writeVarInt(p.hand);
        buf.writeVarInt(p.id);
        buf.writeBoolean(p.flag);
    }

    public static HandsPacket decode(PacketBuffer buf) {
        int action = buf.readVarInt();
        int hand = buf.readVarInt();
        int id = buf.readVarInt();
        boolean flag = buf.readBoolean();
        return new HandsPacket(action, hand, id, flag);
    }

    public static void handle(HandsPacket p, Supplier<NetworkEvent.Context> ctx) {
        ctx.get().enqueueWork(() -> {
            ServerPlayerEntity player = ctx.get().getSender();
            if (player == null) {
                // received on the client
                if (p.action == SET_MODE) {
                    DistExecutor.unsafeRunWhenOn(Dist.CLIENT, () -> () -> ClientHooks.applyMode(p.flag));
                }
                return;
            }
            if (p.action == REQUEST_TOGGLE) {
                if (!player.createCommandSourceStack().hasPermission(2)) {
                    player.sendMessage(new TranslationTextComponent("message.handson.need_cheats"), Util.NIL_UUID);
                    return;
                }
                ServerEvents.setMode(player, !ServerEvents.MODE.contains(player.getUUID()));
                return;
            }
            if (!ServerEvents.MODE.contains(player.getUUID())) {
                return;
            }
            if (!HandSlots.ready(player)) {
                return;
            }
            int h = p.hand == 1 ? 1 : 0;
            if (p.action == GRAB) {
                grab(player, h, p.id);
            } else if (p.action == DROP) {
                drop(player, h);
            } else if (p.action == RING) {
                ringMove(player, h, p.id);
            }
        });
        ctx.get().setPacketHandled(true);
    }

    private static void ringMove(ServerPlayerEntity player, int h, int slot) {
        if (slot < 0 || slot >= HandSlots.RING_COUNT || !HandSlots.ringReady(player)) {
            return;
        }
        ItemStack hand = HandSlots.get(player, h);
        ItemStack in = HandSlots.getRing(player, slot);
        if (hand.isEmpty() && in.isEmpty()) {
            return;
        }
        if (hand.isEmpty()) {
            HandSlots.set(player, h, in.copy());
            HandSlots.setRing(player, slot, ItemStack.EMPTY);
        } else if (in.isEmpty()) {
            HandSlots.setRing(player, slot, hand.copy());
            HandSlots.set(player, h, ItemStack.EMPTY);
        } else if (ItemStack.isSame(hand, in) && ItemStack.tagMatches(hand, in)
                && in.getCount() < in.getMaxStackSize()) {
            int move = Math.min(in.getMaxStackSize() - in.getCount(), hand.getCount());
            in.grow(move);
            hand.shrink(move);
            HandSlots.setRing(player, slot, in);
            HandSlots.set(player, h, hand.isEmpty() ? ItemStack.EMPTY : hand);
        } else {
            ItemStack a = hand.copy();
            ItemStack b = in.copy();
            HandSlots.set(player, h, b);
            HandSlots.setRing(player, slot, a);
        }
    }

    private static void grab(ServerPlayerEntity player, int h, int entityId) {
        Entity e = player.level.getEntity(entityId);
        if (!(e instanceof ItemEntity) || !e.isAlive()) {
            return;
        }
        if (player.distanceToSqr(e) > 36.0D) {
            return;
        }
        ItemEntity itemEntity = (ItemEntity) e;
        ItemStack src = itemEntity.getItem();
        ItemStack cur = HandSlots.get(player, h);
        if (cur.isEmpty()) {
            HandSlots.set(player, h, src.copy());
            itemEntity.remove();
        } else if (ItemStack.isSame(cur, src) && ItemStack.tagMatches(cur, src)) {
            int space = cur.getMaxStackSize() - cur.getCount();
            int move = Math.min(space, src.getCount());
            if (move > 0) {
                cur.grow(move);
                HandSlots.set(player, h, cur);
                src.shrink(move);
                if (src.isEmpty()) {
                    itemEntity.remove();
                } else {
                    itemEntity.setItem(src);
                }
            }
        }
    }

    private static void drop(ServerPlayerEntity player, int h) {
        ItemStack cur = HandSlots.get(player, h);
        if (cur.isEmpty()) {
            return;
        }
        ItemStack dropped = cur.copy();
        HandSlots.set(player, h, ItemStack.EMPTY);
        Vector3d look = player.getLookAngle();
        Vector3d pos = player.getEyePosition(1.0F).add(look.x * 0.5D, look.y * 0.5D - 0.3D, look.z * 0.5D);
        ItemEntity itemEntity = new ItemEntity(player.level, pos.x, pos.y, pos.z, dropped);
        itemEntity.setDeltaMovement(look.x * 0.25D, look.y * 0.25D + 0.1D, look.z * 0.25D);
        itemEntity.setPickUpDelay(20);
        player.level.addFreshEntity(itemEntity);
    }
}

public class ServerEvents {
    /** Players who currently use the Hands On controls. */
    public static final Set<UUID> MODE = ConcurrentHashMap.newKeySet();

    private static final String TAG_UNBREAKABLE = "Unbreakable";
    private static final String TAG_TEMP = "HandsOnTemp";

    /** Sets the control mode for a player and tells the client about it. */
    public static void setMode(ServerPlayerEntity player, boolean on) {
        if (on) {
            MODE.add(player.getUUID());
        } else {
            MODE.remove(player.getUUID());
        }
        HandsNet.CHANNEL.send(PacketDistributor.PLAYER.with(() -> player),
                new HandsPacket(HandsPacket.SET_MODE, 0, 0, on));
    }

    @SubscribeEvent
    public void onRegisterCommands(RegisterCommandsEvent event) {
        ModCommands.register(event.getDispatcher());
    }

    @SubscribeEvent
    public void onPickup(EntityItemPickupEvent event) {
        if (MODE.contains(event.getPlayer().getUUID())) {
            event.setCanceled(true);
        }
    }

    /** Dropped items never despawn: the despawn is cancelled and the timer is reset. */
    @SubscribeEvent
    public void onItemExpire(ItemExpireEvent event) {
        event.setExtraLife(6000);
        event.setCanceled(true);
    }

    /** Adds the hand slots to every player (client and server). */
    @SubscribeEvent
    public void onJoin(EntityJoinWorldEvent event) {
        if (event.getEntity() instanceof PlayerEntity) {
            HandSlots.install((PlayerEntity) event.getEntity());
        }
    }

    /** Safety net in case a player did not get the slots when joining. */
    @SubscribeEvent
    public void onPlayerTickStart(TickEvent.PlayerTickEvent event) {
        if (event.phase == TickEvent.Phase.START) {
            HandSlots.install(event.player);
        }
    }

    /** The hand slots survive death (keepInventory) and dimension changes. */
    @SubscribeEvent
    public void onClone(PlayerEvent.Clone event) {
        CompoundNBT old = event.getOriginal().getPersistentData();
        for (String tag : new String[] {HandsInventory.TAG, RingInventory.TAG}) {
            if (old.contains(tag, 10)) {
                event.getPlayer().getPersistentData().put(tag, old.getCompound(tag).copy());
            }
        }
    }

    /** Items in the hand slots drop on death unless keepInventory is on. */
    @SubscribeEvent
    public void onDeath(LivingDeathEvent event) {
        if (!(event.getEntity() instanceof ServerPlayerEntity)) {
            return;
        }
        ServerPlayerEntity player = (ServerPlayerEntity) event.getEntity();
        if (player.level.getGameRules().getBoolean(GameRules.RULE_KEEPINVENTORY)) {
            return;
        }
        for (int h = 0; h < 2; h++) {
            ItemStack stack = HandSlots.get(player, h);
            if (!stack.isEmpty()) {
                player.drop(stack.copy(), true, false);
                HandSlots.set(player, h, ItemStack.EMPTY);
            }
        }
        for (int i = 0; i < HandSlots.RING_COUNT; i++) {
            ItemStack stack = HandSlots.getRing(player, i);
            if (!stack.isEmpty()) {
                player.drop(stack.copy(), true, false);
                HandSlots.setRing(player, i, ItemStack.EMPTY);
            }
        }
    }

    @SubscribeEvent
    public void onLogout(PlayerEvent.PlayerLoggedOutEvent event) {
        MODE.remove(event.getPlayer().getUUID());
    }

    @SubscribeEvent
    public void onPlayerTick(TickEvent.PlayerTickEvent event) {
        if (event.phase != TickEvent.Phase.END || event.player.level.isClientSide) {
            return;
        }
        if (event.player.tickCount % 10 != 0) {
            return;
        }
        boolean on = MODE.contains(event.player.getUUID());
        PlayerInventory inv = event.player.inventory;
        applyDurability(inv.items, on);
        applyDurability(inv.armor, on);
        applyDurability(inv.offhand, on);
        applyDurability(Arrays.asList(HandSlots.get(event.player, 0), HandSlots.get(event.player, 1)), on);
        ItemStack[] ring = new ItemStack[HandSlots.RING_COUNT];
        for (int i = 0; i < ring.length; i++) {
            ring[i] = HandSlots.getRing(event.player, i);
        }
        applyDurability(Arrays.asList(ring), on);
    }

    /** In Hands On mode damageable items become unbreakable; the marker is removed again afterwards. */
    private static void applyDurability(List<ItemStack> list, boolean on) {
        for (ItemStack stack : list) {
            if (stack.isEmpty()) {
                continue;
            }
            if (on) {
                if (stack.isDamageableItem()) {
                    CompoundNBT tag = stack.getOrCreateTag();
                    if (!tag.getBoolean(TAG_UNBREAKABLE)) {
                        tag.putBoolean(TAG_UNBREAKABLE, true);
                        tag.putBoolean(TAG_TEMP, true);
                    }
                }
            } else {
                CompoundNBT tag = stack.getTag();
                if (tag != null && tag.getBoolean(TAG_TEMP)) {
                    tag.remove(TAG_UNBREAKABLE);
                    tag.remove(TAG_TEMP);
                    if (tag.isEmpty()) {
                        stack.setTag(null);
                    }
                }
            }
        }
    }
}

            HandsState.prevSmoothPitch = HandsState.smoothPitch;
            HandsState.smoothYaw += MathHelper.wrapDegrees(mc.player.yRot - HandsState.smoothYaw) * 0.4F;
            HandsState.smoothPitch += (mc.player.xRot - HandsState.smoothPitch) * 0.4F;
        }

        boolean lockNow = mc.screen == null && HandKeys.LOCK.isDown();
        if (lockNow && !lockWas) {
            HandsState.locked = !HandsState.locked;
        }
        lockWas = lockNow;

        // G / Z open and close the ring inventory
        boolean invNow = mc.screen == null && (HandKeys.INVENTORY.isDown() || HandKeys.INVENTORY_ALT.isDown());
        if (invNow && !invWas) {
            HandsState.invOpen = !HandsState.invOpen;
            if (HandsState.invOpen) {
                HandsState.anchorYaw = mc.player.yRot;
            }
        }
        invWas = invNow;

        for (int h = 0; h < 2; h++) {
            KeyBinding key = h == 0 ? HandKeys.LEFT : HandKeys.RIGHT;
            boolean now = mc.screen == null && key.isDown();
            boolean was = HandsState.down[h];
            ItemEntity prevTarget = HandsState.target[h];
            int prevRing = HandsState.ringHover[h];

            HandsState.prevExt[h] = HandsState.ext[h];
            if (now) {
                HandsState.ext[h] = Math.min(1.0F, HandsState.ext[h] + SPEED);
                int ringSlot = HandsState.invOpen ? findRingSlot(mc) : -1;
                HandsState.ringHover[h] = ringSlot;
                HandsState.target[h] = ringSlot >= 0 ? null : findTarget(mc);
            } else {
                HandsState.ext[h] = Math.max(0.0F, HandsState.ext[h] - SPEED);
                HandsState.ringHover[h] = -1;
                HandsState.target[h] = null;
            }
            if (was && !now) {
                onRelease(mc, h, prevTarget, prevRing);
            }
            HandsState.down[h] = now;
        }
    }

    private void onRelease(Minecraft mc, int h, ItemEntity target, int ringSlot) {
        if (ringSlot >= 0) {
            HandsNet.CHANNEL.sendToServer(new HandsPacket(HandsPacket.RING, h, ringSlot, false));
            return;
        }
        ItemStack held = HandSlots.get(mc.player, h);
        if (!held.isEmpty()) {
            if (!HandsState.locked) {
                HandsNet.CHANNEL.sendToServer(new HandsPacket(HandsPacket.DROP, h, 0, false));
            }
        } else if (target != null && target.isAlive()) {
            HandsNet.CHANNEL.sendToServer(new HandsPacket(HandsPacket.GRAB, h, target.getId(), false));
        }
    }

    private static ItemEntity findTarget(Minecraft mc) {
        ClientPlayerEntity p = mc.player;
        Vector3d start = p.getEyePosition(1.0F);
        Vector3d look = p.getViewVector(1.0F);
        Vector3d end = start.add(look.x * REACH, look.y * REACH, look.z * REACH);
        AxisAlignedBB area = p.getBoundingBox().expandTowards(look.x * REACH, look.y * REACH, look.z * REACH).inflate(1.0D);
        ItemEntity best = null;
        double bestDist = REACH * REACH;
        for (ItemEntity ie : mc.level.getEntitiesOfClass(ItemEntity.class, area)) {
            AxisAlignedBB bb = ie.getBoundingBox().inflate(0.25D);
            double d;
            if (bb.contains(start)) {
                d = 0.0D;
            } else {
                Optional<Vector3d> hit = bb.clip(start, end);
                if (!hit.isPresent()) {
                    continue;
                }
                d = start.distanceToSqr(hit.get());
            }
            if (d < bestDist) {
                bestDist = d;
                best = ie;
            }
        }
        return best;
    }

    @SubscribeEvent
    public void onRenderHand(RenderHandEvent event) {
        if (!HandsState.modeOn) {
            return;

    private static void tr(MatrixStack ms, Pose p, double x, double y, double z) {
        ms.translate(x, y, z);
        p.translate(x, y, z);
    }

    private static void rx(MatrixStack ms, Pose p, double deg) {
        ms.mulPose(net.minecraft.util.math.vector.Vector3f.XP.rotationDegrees((float) deg));
        p.rotX(deg);
    }

    private static void ry(MatrixStack ms, Pose p, double deg) {
        ms.mulPose(net.minecraft.util.math.vector.Vector3f.YP.rotationDegrees((float) deg));
        p.rotY(deg);
    }

    private static void rz(MatrixStack ms, Pose p, double deg) {
        ms.mulPose(net.minecraft.util.math.vector.Vector3f.ZP.rotationDegrees((float) deg));
        p.rotZ(deg);
    }

    private static void sc(MatrixStack ms, Pose p, double x, double y, double z) {
        ms.scale((float) x, (float) y, (float) z);
        p.scale(x, y, z);
    }

    private static void drawHand(Minecraft mc, ClientPlayerEntity player, MatrixStack ms,
                                 IRenderTypeBuffer buf, int light, int h, float pt) {
        boolean right = h == 1;
        double f = right ? 1.0D : -1.0D;
        float ext = MathHelper.lerp(pt, HandsState.prevExt[h], HandsState.ext[h]);

        // the hands only move forward when extended; the only rotation is the lag behind head turns
        double yawRot = 0.0D;
        double pitchRot = 0.0D;
        if (HandsState.smoothInit) {
            float sy = HandsState.prevSmoothYaw
                    + MathHelper.wrapDegrees(HandsState.smoothYaw - HandsState.prevSmoothYaw) * pt;
            float sp = MathHelper.lerp(pt, HandsState.prevSmoothPitch, HandsState.smoothPitch);
            double lagYaw = MathHelper.clamp(MathHelper.wrapDegrees(player.getViewYRot(pt) - sy), -18.0F, 18.0F);
            double lagPitch = MathHelper.clamp(player.getViewXRot(pt) - sp, -18.0F, 18.0F);
            yawRot += lagYaw * LAG_SCALE;
            pitchRot += lagPitch * LAG_SCALE;
        }

        Pose pose = new Pose();
        ms.pushPose();
        ry(ms, pose, yawRot);
        rx(ms, pose, pitchRot);
        tr(ms, pose, 0.0D, 0.0D, -REACH_FWD * ext);

        // resting pose of the arm (the same pose as in 0.1.0)
        tr(ms, pose, f * 0.64D, -0.6D, -0.72D);
        ry(ms, pose, f * 45.0D);
        tr(ms, pose, f * -1.0D, 3.6D, 3.5D);
        rz(ms, pose, f * 120.0D);
        rx(ms, pose, 200.0D);
        ry(ms, pose, f * -135.0D);
        tr(ms, pose, f * 5.6D, 0.0D, 0.0D);
        sc(ms, pose, 1.0D, 1.0D + STRETCH * ext, 1.0D);

        // tip of the arm in the space the hand started in (centre of the end of the arm)
        double[] tip = pose.apply(-f * 0.375D, 0.75D, 0.0D);

        PlayerRenderer renderer = (PlayerRenderer) mc.getEntityRenderDispatcher().getRenderer(player);
        if (right) {
            renderer.renderRightHand(ms, buf, light, player);
        } else {
            renderer.renderLeftHand(ms, buf, light, player);
        }
        ms.popPose();

        // item sits at the tip of the arm and moves with it
        ItemStack stack = HandSlots.get(player, h);
        if (!stack.isEmpty()) {
            ms.pushPose();
            ms.translate(tip[0], tip[1] + ITEM_UP, tip[2] + ITEM_FWD);
            mc.getItemInHandRenderer().renderItem(player, stack,
                    right ? ItemCameraTransforms.TransformType.FIRST_PERSON_RIGHT_HAND
                          : ItemCameraTransforms.TransformType.FIRST_PERSON_LEFT_HAND,
                    !right, ms, buf, light);
            ms.popPose();
        }
    }

    /** Returns {x, y, z, yaw, tilt} of ring slot i. The ring follows the player but keeps its own direction. */
    private static double[] ringSlot(ClientPlayerEntity p, float pt, int i) {
        double px = p.xo + (p.getX() - p.xo) * pt;
        double py = p.yo + (p.getY() - p.yo) * pt;
        double pz = p.zo + (p.getZ() - p.zo) * pt;
        double by = py + p.getEyeHeight() - 0.4D;
        double yaw;
        double dy = 0.0D;
        double r = RING_R;
        double tilt = 0.0D;
        if (i < 8) {
            yaw = HandsState.anchorYaw + i * 45.0D;
        } else {
            int k = (i - 8) % 4;
            yaw = HandsState.anchorYaw + k * 90.0D;
            r = RING_R - 0.15D;
            if (i < 12) {
                dy = 0.62D;
                tilt = -35.0D;
            } else {
                dy = -0.62D;
                tilt = 35.0D;
            }
        }
        double ang = Math.toRadians(yaw);
        return new double[] {px - Math.sin(ang) * r, by + dy, pz + Math.cos(ang) * r, yaw, tilt};
    }

    /** The ring slot the player looks at, or -1. */
    private static int findRingSlot(Minecraft mc) {
        ClientPlayerEntity p = mc.player;
        if (!HandSlots.ringReady(p)) {
            return -1;
        }
        Vector3d eye = p.getEyePosition(1.0F);
        Vector3d look = p.getViewVector(1.0F);
        int best = -1;
        double bestD = 0.3D;
        for (int i = 0; i < HandSlots.RING_COUNT; i++) {
            double[] s = ringSlot(p, 1.0F, i);
            Vector3d v = new Vector3d(s[0] - eye.x, s[1] - eye.y, s[2] - eye.z);
            double t = v.dot(look);
            if (t <= 0.0D) {
                continue;
            }
            double d = v.subtract(look.scale(t)).length();
            if (d < bestD) {
                bestD = d;
                best = i;
            }
        }
        return best;
    }

    private static void ringTransform(MatrixStack ms, double[] s, Vector3d cam) {
        ms.translate(s[0] - cam.x, s[1] - cam.y, s[2] - cam.z);
        ms.mulPose(Vector3f.YP.rotationDegrees((float) -s[3]));
        ms.mulPose(Vector3f.XP.rotationDegrees((float) s[4]));
    }

    private static void drawRing(Minecraft mc, MatrixStack ms, IRenderTypeBuffer.Impl buf, Vector3d cam, float pt) {
        ClientPlayerEntity p = mc.player;
        if (p == null || !HandSlots.ringReady(p)) {
            return;
        }
        // frames (only the outline)
        IVertexBuilder lines = buf.getBuffer(RenderType.lines());
        for (int i = 0; i < HandSlots.RING_COUNT; i++) {
            float r = 0.85F;
            float g = 0.85F;
            float b = 0.85F;
            if (HandsState.ringHover[1] == i) {
                r = 1.0F;
                g = 0.2F;
                b = 0.2F;
            } else if (HandsState.ringHover[0] == i) {
                r = 0.25F;
                g = 0.5F;
                b = 1.0F;
            }
            double[] s = ringSlot(p, pt, i);
            ms.pushPose();
            ringTransform(ms, s, cam);
            WorldRenderer.renderLineBox(ms, lines, -PANEL, -PANEL, -0.01D, PANEL, PANEL, 0.01D, r, g, b, 0.9F);
            WorldRenderer.renderLineBox(ms, lines, -PANEL - 0.01D, -PANEL - 0.01D, -0.01D,
                    PANEL + 0.01D, PANEL + 0.01D, 0.01D, r, g, b, 0.9F);
            ms.popPose();
        }
        buf.endBatch(RenderType.lines());

        // items and counts
        for (int i = 0; i < HandSlots.RING_COUNT; i++) {
            ItemStack st = HandSlots.getRing(p, i);
            if (st.isEmpty()) {
                continue;
            }
            double[] s = ringSlot(p, pt, i);
            ms.pushPose();
            ringTransform(ms, s, cam);
            ms.mulPose(Vector3f.YP.rotationDegrees(180.0F));
            ms.scale(0.4F, 0.4F, 0.4F);
            mc.getItemRenderer().renderStatic(st, ItemCameraTransforms.TransformType.FIXED,
                    15728880, OverlayTexture.NO_OVERLAY, ms, buf);
            ms.popPose();

            if (st.getCount() > 1) {
                String text = String.valueOf(st.getCount());
                ms.pushPose();
                ms.translate(s[0] - cam.x, s[1] - cam.y - 0.22D, s[2] - cam.z);
                ms.mulPose(mc.getEntityRenderDispatcher().cameraOrientation());
                ms.scale(-0.02F, -0.02F, 0.02F);
                float w = mc.font.width(text);
                mc.font.drawInBatch(text, -w / 2.0F, 0.0F, 0xFFFFFFFF, false,
                        ms.last().pose(), buf, true, 0, 15728880);
                ms.popPose();
            }
        }
        buf.endBatch();
    }

    @SubscribeEvent
    public void onRenderWorld(RenderWorldLastEvent event) {
        if (!HandsState.modeOn) {
            return;
        }
        Minecraft mc = Minecraft.getInstance();
        if (mc.level == null) {
            return;
        }
        MatrixStack ms = event.getMatrixStack();
        Vector3d cam = mc.gameRenderer.getMainCamera().getPosition();
        IRenderTypeBuffer.Impl buf = mc.renderBuffers().bufferSource();

        if (HandsState.invOpen) {
            drawRing(mc, ms, buf, cam, event.getPartialTicks());
        }

        for (int h = 0; h < 2; h++) {
            ItemEntity t = HandsState.target[h];
            if (!HandsState.down[h] || t == null || !t.isAlive()) {
                continue;
            }
            // left hand = blue, right hand = red
            float r = h == 0 ? 0.25F : 1.0F;
            float g = h == 0 ? 0.5F : 0.2F;
            float b = h == 0 ? 1.0F : 0.2F;

            AxisAlignedBB box = t.getBoundingBox().inflate(0.08D + 0.03D * h);
            IVertexBuilder lines = buf.getBuffer(RenderType.lines());
            ms.pushPose();
            ms.translate(-cam.x, -cam.y, -cam.z);
            WorldRenderer.renderLineBox(ms, lines, box, r, g, b, 1.0F);
            ms.popPose();
            buf.endBatch(RenderType.lines());

            int count = t.getItem().getCount();
            if (count > 1) {
                String text = String.valueOf(count);
                ms.pushPose();
                ms.translate(t.getX() - cam.x, t.getY() + t.getBbHeight() + 0.4D - cam.y, t.getZ() - cam.z);
                ms.mulPose(mc.getEntityRenderDispatcher().cameraOrientation());
                ms.scale(-0.03F, -0.03F, 0.03F);
                float w = mc.font.width(text);
                int color = (255 << 24) | ((int) (r * 255) << 16) | ((int) (g * 255) << 8) | (int) (b * 255);
                mc.font.drawInBatch(text, -w / 2.0F + (h == 0 ? -8.0F : 8.0F), 0.0F, color, false,
                        ms.last().pose(), buf, true, 0, 15728880);
                ms.popPose();
                buf.endBatch();
            }
        }
    }

    private static boolean isLowered(RenderGameOverlayEvent.ElementType t) {
        return t == RenderGameOverlayEvent.ElementType.HEALTH
                || t == RenderGameOverlayEvent.ElementType.ARMOR
                || t == RenderGameOverlayEvent.ElementType.FOOD
                || t == RenderGameOverlayEvent.ElementType.AIR
                || t == RenderGameOverlayEvent.ElementType.HEALTHMOUNT
                || t == RenderGameOverlayEvent.ElementType.EXPERIENCE
                || t == RenderGameOverlayEvent.ElementType.JUMPBAR;
    }

    /** The hotbar is hidden; health, food, armor and experience move down to where it was. */
    @SubscribeEvent
    public void onOverlayPre(RenderGameOverlayEvent.Pre event) {
        if (!HandsState.modeOn) {
            return;
        }
        RenderGameOverlayEvent.ElementType t = event.getType();
        if (t == RenderGameOverlayEvent.ElementType.HOTBAR) {
            event.setCanceled(true);
        } else if (isLowered(t) && !event.isCanceled()) {
            event.getMatrixStack().pushPose();
            event.getMatrixStack().translate(0.0D, 22.0D, 0.0D);
        }
    }

    public void onOverlayPostLowered(RenderGameOverlayEvent.Post event) {
        if (HandsState.modeOn && isLowered(event.getType())) {
            event.getMatrixStack().popPose();
        }
    }

    private static String tr(String key, Object... args) {
        return new TranslationTextComponent(key, args).getString();

    /** Small hints in the bottom left corner. Shift shows the extended versions. */
    private static void drawHints(Minecraft mc, MatrixStack ms, int hgt) {
        boolean more = Screen.hasShiftDown();
        List<String> lines = new ArrayList<>();

        boolean ground = false;
        boolean slot = false;
        for (int h = 0; h < 2; h++) {
            if (HandsState.target[h] != null) {
                ground = true;
            }
            if (HandsState.ringHover[h] >= 0) {
                slot = true;
            }
        }
        if (slot) {
            lines.add(tr("hint.handson.slot"));
        } else if (ground) {
            lines.add(tr("hint.handson.take"));
        }

        lines.add(tr("hint.handson.hands", HandKeys.LEFT.getTranslatedKeyMessage(),
                HandKeys.RIGHT.getTranslatedKeyMessage()) + (more ? " " + tr("hint.handson.hands.more") : ""));
        lines.add(tr("hint.handson.lock", HandKeys.LOCK.getTranslatedKeyMessage())
                + (more ? " " + tr("hint.handson.lock.more") : ""));
        lines.add(tr("hint.handson.inventory", HandKeys.INVENTORY.getTranslatedKeyMessage(),
                HandKeys.INVENTORY_ALT.getTranslatedKeyMessage())
                + (more ? " " + tr("hint.handson.inventory.more") : ""));

        float scale = 0.7F;
        ms.pushPose();
        ms.scale(scale, scale, 1.0F);
        float y = (hgt - 5) / scale - 9.0F * lines.size();
        for (String line : lines) {
            mc.font.drawShadow(ms, line, 6.0F / scale, y, 0xBBBBBB);
            y += 9.0F;
        }
        ms.popPose();
    }

    private static void drawHeld(Minecraft mc, MatrixStack ms, ItemStack stack, int x, int y, boolean right) {
        drawHotbarCell(mc, ms, x, y);
        if (stack.isEmpty()) {
            return;
        }
        mc.getItemRenderer().renderAndDecorateItem(stack, x, y);
        mc.getItemRenderer().renderGuiItemDecorations(mc.font, stack, x, y);

        // the name above the slot, wrapped into up to 5 lines
        List<IReorderingProcessor> lines = mc.font.split(stack.getHoverName(), 90);
        int n = Math.min(lines.size(), 5);
        for (int i = 0; i < n; i++) {
